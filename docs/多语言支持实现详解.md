## 📖 引言

国际化（Internationalization，简称 i18n）是现代应用必备的功能，它允许应用支持多种语言，满足不同地区用户的需求。对于 Electron 桌面应用来说，实现多语言支持不仅能提升用户体验，还能扩大应用的受众范围。

### 国际化的应用场景

- **全球化的桌面应用**：面向全球用户的应用
- **企业级应用**：跨国企业内部使用的工具
- **开发者工具**：支持多语言的 IDE 和编辑器
- **生产力工具**：面向多语言用户的办公软件

### 本文将实现的功能

✅ 自动检测系统语言  
✅ 支持多种语言切换（简体中文、英语、日语、韩语）  
✅ 翻译资源管理和加载  
✅ 参数化翻译支持  
✅ 语言切换实时生效  
✅ 数字、日期、货币格式化  
✅ 鸿蒙平台适配

## 💻 代码实现

### 1. 创建 i18n 模块

创建 `app/i18n.js` 文件，实现完整的国际化功能：

```javascript
const fs = require('fs');
const path = require('path');
const { app } = require('electron');

// 当前语言
let currentLanguage = 'zh-CN';

// 语言缓存
let translationsCache = {};

// 支持的语言列表
const supportedLanguages = [
    { code: 'zh-CN', name: '简体中文', nativeName: '简体中文' },
    { code: 'en-US', name: 'English', nativeName: 'English' },
    { code: 'ja-JP', name: '日本語', nativeName: '日本語' },
    { code: 'ko-KR', name: '한국어', nativeName: '한국어' }
];

/**
 * 初始化 i18n 模块
 */
function initI18n() {
    // 检测系统语言
    const systemLanguage = detectSystemLanguage();

    // 设置当前语言
    setLanguage(systemLanguage);

    console.log('i18n 模块初始化完成，当前语言:', currentLanguage);
}
```

### 2. 检测系统语言

```javascript
/**
 * 检测系统语言
 */
function detectSystemLanguage() {
    // 方法1: 从 app.getLocale() 获取
    try {
        const locale = app.getLocale();
        console.log('系统语言 (getLocale):', locale);

        if (isLanguageSupported(locale)) {
            return locale;
        }

        // 尝试匹配语言代码的前两部分
        const langCode = locale.split('-')[0];
        const matchedLang = supportedLanguages.find(lang => lang.code.startsWith(langCode));
        if (matchedLang) {
            return matchedLang.code;
        }
    } catch (error) {
        console.warn('无法获取系统语言:', error);
    }

    // 方法2: 从环境变量获取
    try {
        const envLang = process.env.LANG || process.env.LC_ALL || process.env.LANGUAGE;
        if (envLang) {
            const langCode = envLang.split('.')[0].replace('_', '-');
            if (isLanguageSupported(langCode)) {
                return langCode;
            }
        }
    } catch (error) {
        console.warn('无法获取系统语言:', error);
    }

    // 使用默认语言
    return 'zh-CN';
}
```

### 3. 加载语言包

```javascript
/**
 * 加载语言包
 */
function loadLanguagePack(languageCode) {
    try {
        // 尝试从文件加载
        const langFilePath = getLanguageFilePath(languageCode);
        if (fs.existsSync(langFilePath)) {
            const langData = fs.readFileSync(langFilePath, 'utf-8');
            translationsCache[languageCode] = JSON.parse(langData);
            console.log(`语言包加载成功: ${languageCode}`);
            return;
        }
    } catch (error) {
        console.warn(`无法从文件加载语言包:`, error);
    }

    // 使用内置的默认翻译
    translationsCache[languageCode] = getDefaultTranslations(languageCode);
    console.log(`使用内置翻译: ${languageCode}`);
}

/**
 * 获取语言文件路径
 */
function getLanguageFilePath(languageCode) {
    const possiblePaths = [
        path.join(__dirname, 'locales', `${languageCode}.json`),
        path.join(__dirname, '..', 'locales', `${languageCode}.json`),
        path.join(__dirname, 'i18n', `${languageCode}.json`)
    ];

    for (const filePath of possiblePaths) {
        if (fs.existsSync(filePath)) {
            return filePath;
        }
    }

    return null;
}
```

### 4. 翻译函数

```javascript
/**
 * 获取翻译文本
 * @param {string} key 翻译键
 * @param {Object} params 参数对象（可选）
 * @returns {string} 翻译后的文本
 */
function t(key, params = {}) {
    // 获取当前语言的翻译
    const translations = translationsCache[currentLanguage];

    // 如果没有找到翻译，尝试使用默认语言
    if (!translations || !translations[key]) {
        console.warn(`翻译键未找到: ${key} (语言: ${currentLanguage})`);

        // 尝试使用简体中文作为后备
        const defaultTranslations = translationsCache['zh-CN'];
        if (defaultTranslations && defaultTranslations[key]) {
            return replaceParams(defaultTranslations[key], params);
        }

        // 如果都没有，返回键本身
        return replaceParams(key, params);
    }

    // 替换参数
    return replaceParams(translations[key], params);
}

/**
 * 替换翻译中的参数
 */
function replaceParams(text, params) {
    if (!params || Object.keys(params).length === 0) {
        return text;
    }

    return text.replace(/\{(\w+)\}/g, (match, key) => {
        return params[key] !== undefined ? params[key] : match;
    });
}
```

### 5. 格式化函数

```javascript
/**
 * 格式化数字（根据语言环境）
 */
function formatNumber(number) {
    try {
        return new Intl.NumberFormat(currentLanguage).format(number);
    } catch (error) {
        console.warn('数字格式化失败:', error);
        return number.toString();
    }
}

/**
 * 格式化日期（根据语言环境）
 */
function formatDate(date, options = {}) {
    try {
        const defaultOptions = {
            year: 'numeric',
            month: 'long',
            day: 'numeric'
        };
        return new Intl.DateTimeFormat(currentLanguage, { ...defaultOptions, ...options }).format(date);
    } catch (error) {
        console.warn('日期格式化失败:', error);
        return date.toString();
    }
}

/**
 * 格式化货币（根据语言环境）
 */
function formatCurrency(amount, currency = 'CNY') {
    try {
        return new Intl.NumberFormat(currentLanguage, {
            style: 'currency',
            currency: currency
        }).format(amount);
    } catch (error) {
        console.warn('货币格式化失败:', error);
        return amount.toString();
    }
}
```

### 7. 在主进程中集成 i18n

在 `main.js` 中集成国际化功能：

```javascript
const i18n = require('./i18n');

// 初始化 i18n
i18n.initI18n();

// 创建多语言菜单
function createMenu() {
    const template = [
        {
            label: i18n.t('menu.file'),
            submenu: [
                {
                    label: i18n.t('button.export'),
                    click: () => {
                        // 导出功能
                    }
                }
            ]
        },
        {
            label: i18n.t('menu.edit'),
            submenu: [
                { role: 'undo' },
                { role: 'redo' }
            ]
        },
        {
            label: i18n.t('menu.view'),
            submenu: [
                { role: 'reload' },
                { role: 'toggleDevTools' }
            ]
        },
        {
            label: i18n.t('menu.help'),
            submenu: [
                {
                    label: i18n.t('settings.language'),
                    submenu: i18n.createLanguageMenuTemplate()
                }
            ]
        }
    ];

    const menu = Menu.buildFromTemplate(template);
    Menu.setApplicationMenu(menu);
}

// 监听语言变化
ipcMain.on('set-language', (event, languageCode) => {
    i18n.setLanguage(languageCode);
    createMenu(); // 重新创建菜单

    // 通知所有窗口语言已更改
    BrowserWindow.getAllWindows().forEach(window => {
        window.webContents.send('language-changed', languageCode);
    });
});
```

## 🌏 鸿蒙平台适配

### 鸿蒙平台的特殊注意事项

#### 1. 语言代码映射

鸿蒙平台可能使用不同的语言代码格式，需要进行映射：

```javascript
/**
 * 鸿蒙平台语言代码映射
 */
const harmonyosLanguageMap = {
    'zh-Hans': 'zh-CN',
    'zh-Hant': 'zh-TW',
    'en': 'en-US',
    'ja': 'ja-JP',
    'ko': 'ko-KR'
};

/**
 * 获取鸿蒙平台语言代码
 */
function getHarmonyOSLanguageCode() {
    try {
        // 鸿蒙平台特有的 API
        const locale = app.getLocale();

        // 映射到标准语言代码
        return harmonyosLanguageMap[locale] || locale;
    } catch (error) {
        console.warn('鸿蒙平台语言检测失败:', error);
        return 'zh-CN';
    }
}
```

#### 2. 字体支持

不同语言需要不同的字体支持：

```javascript
/**
 * 获取语言对应的字体
 */
function getLanguageFont(languageCode) {
    const fontMap = {
        'zh-CN': 'PingFang SC, Microsoft YaHei, sans-serif',
        'en-US': 'Segoe UI, Helvetica Neue, Arial, sans-serif',
        'ja-JP': 'Hiragino Sans, Yu Gothic, Meiryo, sans-serif',
        'ko-KR': 'Malgun Gothic, Apple SD Gothic Neo, sans-serif'
    };

    return fontMap[languageCode] || 'sans-serif';
}
```

#### 3. 布局方向

某些语言（如阿拉伯语、希伯来语）需要从右到左的布局：

```javascript
/**
 * 获取语言布局方向
 */
function getLanguageDirection(languageCode) {
    const rtlLanguages = ['ar', 'he', 'fa', 'ur'];
    const langCode = languageCode.split('-')[0];

    return rtlLanguages.includes(langCode) ? 'rtl' : 'ltr';
}
```

### 2. 翻译管理工具

```javascript
/**
 * 提取代码中未翻译的文本
 */
function extractUntranslatedText() {
    const glob = require('glob');
    const files = glob.sync('app/**/*.{js,html}');

    const translations = getAllTranslations();
    const untranslated = [];

    files.forEach(file => {
        const content = fs.readFileSync(file, 'utf-8');
        const matches = content.match(/t\(['"]([^'"]+)['"]\)/g);

        if (matches) {
            matches.forEach(match => {
                const key = match.match(/t\(['"]([^'"]+)['"]\)/)[1];
                if (!translations[key]) {
                    untranslated.push({ key, file });
                }
            });
        }
    });

    return untranslated;
}
```

### 3. 翻译贡献工具

```javascript
/**
 * 生成翻译贡献模板
 */
function generateTranslationTemplate(sourceLang, targetLang) {
    const sourceTranslations = getAllTranslations(sourceLang);
    const template = {};

    Object.keys(sourceTranslations).forEach(key => {
        template[key] = ''; // 留空供翻译
    });

    const outputPath = path.join(__dirname, 'locales', `${targetLang}.json`);
    fs.writeFileSync(outputPath, JSON.stringify(template, null, 2), 'utf-8');

    console.log(`翻译模板已生成: ${outputPath}`);
}
```

## 📊 性能优化

### 1. 翻译缓存

```javascript
// ✅ 优化：缓存翻译结果
const translationCache = new Map();

function getCachedTranslation(key, params) {
    const cacheKey = `${currentLanguage}:${key}:${JSON.stringify(params)}`;

    if (translationCache.has(cacheKey)) {
        return translationCache.get(cacheKey);
    }

    const result = t(key, params);
    translationCache.set(cacheKey, result);

    return result;
}
```

## 🐛 常见问题与解决方案

### 问题 1：翻译键未找到

**原因**：
- 翻译键拼写错误
- 翻译文件未加载
- 语言代码不匹配

**解决方案**：
```javascript
// 添加详细的日志
function t(key, params) {
    if (!translationsCache[currentLanguage]) {
        console.error(`语言包未加载: ${currentLanguage}`);
    }

    if (!translationsCache[currentLanguage][key]) {
        console.warn(`翻译键未找到: ${key}`);
    }

    // 返回后备翻译
}
```

### 问题 2：语言切换不生效

**原因**：
- 未通知渲染进程
- 未重新创建菜单
- 缓存未清除

**解决方案**：
```javascript
function setLanguage(languageCode) {
    currentLanguage = languageCode;
    translationsCache = {}; // 清除缓存

    // 通知所有窗口
    BrowserWindow.getAllWindows().forEach(window => {
        window.webContents.send('language-changed', languageCode);
    });

    // 重新创建菜单
    createMenu();
}
```

### 问题 3：鸿蒙平台语言检测异常

**原因**：
- 鸿蒙平台的 API 兼容性问题
- 语言代码格式不同

**解决方案**：
```javascript
function detectSystemLanguage() {
    try {
        // 尝试鸿蒙平台特有的方法
        const locale = app.getLocale();

        // 映射语言代码
        if (harmonyosLanguageMap[locale]) {
            return harmonyosLanguageMap[locale];
        }

        return locale;
    } catch (error) {
        console.warn('鸿蒙平台语言检测失败:', error);
        return 'zh-CN';
    }
}
```
