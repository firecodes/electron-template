function createTray() {
    const iconPath = getTrayIconPath();
    tray = new Tray(iconPath);
    tray.setToolTip('HarmonyOS Electron 系统信息查看器');
    tray.setContextMenu(createContextMenu());
    tray.on('click', toggleMainWindow);
}

function showTrayNotification(title, body) {
    const notification = new Notification({
        title: title,
        body: body,
        icon: getTrayIconPath()

    notification.show();

#### 核心代码示例

```javascript
// 检查更新
async function checkForUpdates(showNotification = true) {
    const releaseInfo = await fetchLatestRelease();
    const latestVersion = releaseInfo.tag_name.replace('v', '');
    const hasUpdate = compareVersions(currentVersion, latestVersion) < 0;

    if (hasUpdate) {
        notifyUpdateAvailable(releaseInfo, showNotification);
    }

    return { hasUpdate, latestVersion };
}

// 下载更新
async function downloadUpdate(version, onProgress) {
    const downloadUrl = getDownloadUrl(version);
    const filePath = await downloadFile(downloadUrl, onProgress);
    return filePath;
}

// 安装更新
async function installUpdate(filePath) {
    if (process.platform === 'harmonyos') {
        await installHarmonyOSUpdate(filePath);
    } else {
        await installGenericUpdate(filePath);
    }
}
```

## 🚀 使用指南

### 集成到项目

#### 1. 复制功能模块

将功能模块文件复制到项目的 `app` 目录：

```bash
cp tray.js /path/to/your/app/
cp i18n.js /path/to/your/app/
cp updater.js /path/to/your/app/
```

#### 2. 在 main.js 中引入

```javascript
const tray = require('./tray');
const i18n = require('./i18n');
const updater = require('./updater');

// 初始化模块
i18n.initI18n();
tray.createTray();
updater.initUpdater();
```

#### 3. 配置菜单

```javascript
function createMenu() {
    const template = [
        {
            label: i18n.t('menu.file'),
            submenu: [
                {
                    label: i18n.t('button.export'),
                    click: () => { /* 导出功能 */ }
                },
                {
                    label: i18n.t('menu.check_update'),
                    click: () => {
                        updater.checkForUpdates(true);
                    }
                }
            ]
        },
        {
            label: i18n.t('settings.language'),
            submenu: i18n.createLanguageMenuTemplate()
        }
    ];

    const menu = Menu.buildFromTemplate(template);
    Menu.setApplicationMenu(menu);
}
```

#### 4. IPC 通信

```javascript
// 主进程
ipcMain.handle('check-for-updates', async () => {
    return await updater.checkForUpdates(true);
});

// 渲染进程
async function checkUpdates() {
    const result = await ipcRenderer.invoke('check-for-updates');
    if (result.hasUpdate) {
        showUpdateDialog(result);
    }
}
```
