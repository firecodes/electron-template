## 📖 引言

系统托盘（System Tray）是桌面应用中一个非常重要的功能，它允许应用在后台运行，并在系统状态栏中显示一个图标。用户可以通过托盘图标快速访问应用功能，而无需打开完整的应用窗口。

### 系统托盘的应用场景

- **后台监控应用**：如系统监控工具、下载管理器
- **即时通讯工具**：如微信、QQ、钉钉
- **通知提醒应用**：如邮件客户端、日程提醒
- **工具类应用**：如截图工具、翻译工具

### 本文将实现的功能

✅ 创建系统托盘图标  
✅ 托盘右键菜单（显示/隐藏窗口、退出应用）  
✅ 托盘单击/双击事件处理  
✅ 托盘通知功能  
✅ 平台兼容性处理（鸿蒙 PC）  
✅ 完整的错误处理和日志记录

## 🔧 技术方案

### Electron Tray API 核心概念

Electron 提供了 `Tray` 模块来管理系统托盘，主要包含以下 API：

| API | 说明 |
|-----|------|
| `new Tray(image)` | 创建托盘图标 |
| `tray.setToolTip(text)` | 设置工具提示 |
| `tray.setContextMenu(menu)` | 设置右键菜单 |
| `tray.on('click', callback)` | 监听点击事件 |
| `tray.on('double-click', callback)` | 监听双击事件 |
| `tray.destroy()` | 销毁托盘 |

## 💻 代码实现

### 1. 创建托盘模块文件

创建 `app/tray.js` 文件，实现完整的托盘功能：

```javascript
const { Tray, Menu, app, nativeImage } = require('electron');
const path = require('path');

let tray = null;

/**
 * 创建系统托盘
 */
function createTray() {
    // 如果托盘已存在，先销毁
    if (tray) {
        tray.destroy();
    }

    // 获取托盘图标路径
    const iconPath = getTrayIconPath();

    // 创建托盘
    tray = new Tray(iconPath);

    // 设置托盘工具提示
    tray.setToolTip('HarmonyOS Electron 系统信息查看器');

    // 创建托盘菜单
    const contextMenu = Menu.buildFromTemplate([
        {
            label: '显示主窗口',
            click: () => {
                showMainWindow();
            }
        },
        {
            label: '隐藏主窗口',
            click: () => {
                hideMainWindow();
            }
        },
        { type: 'separator' },
        {
            label: '系统信息',
            click: () => {
                showMainWindow();
            }
        },
        {
            label: '退出应用',
            click: () => {
                quitApp();
            }
        }
    ]);

    // 设置托盘菜单
    tray.setContextMenu(contextMenu);

    // 托盘点击事件
    tray.on('click', () => {
        console.log('托盘被点击');
        toggleMainWindow();
    });

    console.log('系统托盘创建成功');
}
```

### 2. 处理图标文件

```javascript
/**
 * 获取托盘图标路径
 */
function getTrayIconPath() {
    // 尝试多个可能的图标路径
    const possiblePaths = [
        path.join(__dirname, 'tray-icon.png'),
        path.join(__dirname, 'electron_white.png'),
        path.join(__dirname, 'icons', 'tray.png')
    ];

    for (const iconPath of possiblePaths) {
        try {
            const fs = require('fs');
            if (fs.existsSync(iconPath)) {
                console.log('使用托盘图标:', iconPath);
                return iconPath;
            }
        } catch (error) {
            continue;
        }
    }

    // 如果找不到图标，创建默认图标
    return createDefaultIcon();
}

/**
 * 创建默认托盘图标
 */
function createDefaultIcon() {
    const size = 16;
    const buffer = Buffer.alloc(size * size * 4);

    // 填充蓝色背景
    for (let i = 0; i < buffer.length; i += 4) {
        buffer[i] = 0;     // R
        buffer[i + 1] = 122; // G
        buffer[i + 2] = 204; // B
        buffer[i + 3] = 255; // A
    }

    const image = nativeImage.createFromBuffer(buffer, {
        width: size,
        height: size
    });

    return image.resize({
        width: 16,
        height: 16,
        quality: 'best'
    });
}
```

### 3. 窗口管理功能

```javascript
/**
 * 显示主窗口
 */
function showMainWindow() {
    if (global.mainWindow) {
        if (global.mainWindow.isMinimized()) {
            global.mainWindow.restore();
        }
        global.mainWindow.show();
        global.mainWindow.focus();
    }
}

/**
 * 隐藏主窗口
 */
function hideMainWindow() {
    if (global.mainWindow) {
        global.mainWindow.hide();
    }
}

/**
 * 切换主窗口显示状态
 */
function toggleMainWindow() {
    if (global.mainWindow) {
        if (global.mainWindow.isVisible()) {
            hideMainWindow();
        } else {
            showMainWindow();
        }
    }
}
```

### 4. 托盘通知功能

```javascript
/**
 * 显示托盘通知
 */
function showTrayNotification(title, body) {
    if (!tray) {
        return;
    }

    const { Notification } = require('electron');

    if (Notification.isSupported()) {
        const notification = new Notification({
            title: title,
            body: body,
            icon: getTrayIconPath(),
            silent: false
        });

        notification.show();

        // 点击通知时显示主窗口
        notification.on('click', () => {
            showMainWindow();
        });
    }
}
```

### 5. 在主进程中集成托盘

在 `main.js` 中集成托盘功能：

```javascript
const { app, BrowserWindow, ipcMain } = require('electron');
const tray = require('./tray');

// 保存主窗口引用到全局
let mainWindow;

function createWindow() {
    mainWindow = new BrowserWindow({
        width: 1200,
        height: 800,
        webPreferences: {
            nodeIntegration: true,
            contextIsolation: false
        }
    });

    mainWindow.loadFile('system-info.html');

    // 保存到全局，供托盘模块使用
    global.mainWindow = mainWindow;

    // 窗口关闭时隐藏而不是退出（可选）
    mainWindow.on('close', (event) => {
        if (!app.isQuitting) {
            event.preventDefault();
            mainWindow.hide();
        }
    });
}

app.whenReady().then(() => {
    createWindow();

    // 创建系统托盘
    tray.createTray();
});

// 应用退出前清理托盘
app.on('before-quit', () => {
    app.isQuitting = true;
    tray.destroyTray();
});
```

## 🌏 鸿蒙平台适配

### 鸿蒙平台的特殊注意事项

#### 1. 图标尺寸要求

不同平台对托盘图标尺寸有不同要求：

| 平台 | 推荐尺寸 | 备注 |
|------|----------|------|
| Windows | 16x16 | 支持 PNG、ICO |
| macOS | 16x16 (1x) / 32x32 (2x) | 支持 PNG、ICNS |
| Linux | 22x22 或 24x24 | 支持 PNG、SVG |
| HarmonyOS | 16x16 | 支持 PNG |

#### 2. 图标透明度处理

```javascript
/**
 * 处理图标透明度（鸿蒙平台兼容）
 */
function processIconForPlatform(image) {
    if (process.platform === 'darwin') {
        // macOS 需要特殊的模板图标
        return image.setTemplateImage(true);
    } else if (process.platform === 'win32') {
        // Windows 可能需要调整透明度
        return image;
    } else {
        // Linux 和 HarmonyOS
        return image;
    }
}
```

#### 3. 鸿蒙平台特有的 API 兼容性

```javascript
/**
 * 创建托盘（鸿蒙平台适配）
 */
function createTray() {
    try {
        const iconPath = getTrayIconPath();
        tray = new Tray(iconPath);

        // 鸿蒙平台可能不支持某些特性
        try {
            tray.setToolTip('HarmonyOS Electron');
        } catch (error) {
            console.warn('鸿蒙平台不支持 setToolTip:', error);
        }

        return true;
    } catch (error) {
        console.error('创建托盘失败:', error);
        return false;
    }
}
```

### 1. 托盘闪烁提醒

```javascript
/**
 * 设置托盘闪烁
 */
function setTrayFlash(flash) {
    if (!tray) return;

    if (flash) {
        // 定时切换图标实现闪烁效果
        let flashInterval = setInterval(() => {
            const isNormal = tray.getImage() === normalIcon;
            tray.setImage(isNormal ? alertIcon : normalIcon);
        }, 500);

        // 保存定时器引用
        tray._flashInterval = flashInterval;
    } else {
        // 停止闪烁
        if (tray._flashInterval) {
            clearInterval(tray._flashInterval);
            tray.setImage(normalIcon);
        }
    }
}
```

## 🐛 常见问题与解决方案

### 问题 1：托盘图标不显示

**原因**：
- 图标路径错误
- 图标文件不存在
- 图标格式不支持

**解决方案**：
```javascript
// 检查图标文件是否存在
const fs = require('fs');
if (!fs.existsSync(iconPath)) {
    console.error('图标文件不存在:', iconPath);
    // 使用默认图标
    iconPath = createDefaultIcon();
}
```

### 问题 2：托盘点击事件不响应

**原因**：
- 事件监听器未正确绑定
- 托盘被销毁后仍尝试访问

**解决方案**：
```javascript
// 确保在创建托盘后绑定事件
if (tray) {
    tray.on('click', () => {
        // 处理点击事件
    });
}
```

### 问题 3：鸿蒙平台托盘功能异常

**原因**：
- 鸿蒙平台的 API 兼容性问题
- 图标尺寸不符合要求

**解决方案**：
```javascript
// 添加平台检测和兼容性处理
if (process.platform === 'harmonyos') {
    // 使用鸿蒙平台特定的配置
    const icon = nativeImage.createFromPath(iconPath);
    const resizedIcon = icon.resize({ width: 16, height: 16 });
    tray = new Tray(resizedIcon);
}
```
